        databaseURL: "https://ВСТАВТЕ-default-rtdb.firebaseio.com",
        projectId: "ВСТАВТЕ",

        messagingSenderId: "ВСТАВТЕ",
        appId: "ВСТАВТЕ"
    };

    const app = initializeApp(firebaseConfig);
    const db = getDatabase(app);

    const pts = [0, 15, 30, 40, "AD"];
    let lastAlertKey = "";
    let forceShowStats = false;
    let statsTimeout = null;

    function updateStyle(p, v) { document.documentElement.style.setProperty(p, v); }

    function renderOverlay(d) {
        if (!d) return;
        const state = d.state || {};
        const ui = d.ui || {};
        const scores = state.scores || [0,0];
        const games = state.games || [0,0];
        const setsHistory = state.setsHistory || [[],[]];
        const server = state.server || 0;
        const isGameActive = state.isGameActive || false;
        const isTiebreak = state.isTiebreak || false;
        const isAmerican = state.isAmerican || false;
        const setTimes = state.setTimes || [0,0,0];
        const activeSetIndex = state.activeSetIndex || 0;
        const notifText = state.notifText || "";
        const showFinish = state.showFinish || false;
        const showIntro = state.showIntro || false;
        const showLogo = state.showLogo || false;
        const showBoard = state.showBoard !== false;
        const isProset = state.isProset || false;
        const prosetLimit = state.prosetLimit || 8;
        const p1name = state.p1name || "Команда А";
        const p2name = state.p2name || "Команда Б";
        const courtName = state.courtName || "COURT 1";

        // Apply UI styles
        if (ui.accentBlue) updateStyle('--accent-blue', ui.accentBlue);
        if (ui.ballColor) updateStyle('--ball-color', ui.ballColor);
        if (ui.panelBg) updateStyle('--panel-bg', ui.panelBg);
        if (ui.drawerBg) updateStyle('--drawer-bg', ui.drawerBg);
        if (ui.popupBg) updateStyle('--popup-bg', ui.popupBg);
        if (ui.popupText) updateStyle('--popup-text', ui.popupText);
        if (ui.teamNameColor) updateStyle('--team-name-color', ui.teamNameColor);
        if (ui.scoreColor) updateStyle('--score-num-color', ui.scoreColor);
        if (ui.finishBg) updateStyle('--finish-bg', ui.finishBg);
        if (ui.fontFamily) updateStyle('--team-font-family', ui.fontFamily);
        if (ui.sbW) updateStyle('--sb-w', ui.sbW);
        if (ui.sbH) updateStyle('--sb-h', ui.sbH);
        if (ui.sbRadius) updateStyle('--sb-radius', ui.sbRadius);
        if (ui.sbSkew) updateStyle('--sb-skew', ui.sbSkew);
        if (ui.logoW) updateStyle('--logo-w', ui.logoW);
        if (ui.logoH) updateStyle('--logo-h', ui.logoH);

        // Scoreboard position
        if (ui.sbTop !== undefined) document.getElementById('scoreboard-anchor').style.top = ui.sbTop + 'px';
        if (ui.sbLeft !== undefined) document.getElementById('scoreboard-anchor').style.left = ui.sbLeft + 'px';

        // Names
        document.getElementById('p1-display').innerText = p1name;
        document.getElementById('p2-display').innerText = p2name;
        document.getElementById('intro-p1-display').innerText = p1name;
        document.getElementById('intro-p2-display').innerText = p2name;
        document.getElementById('intro-court-display').innerText = courtName;

        // Score display
        const getL = (i) => {
            if (isAmerican || isTiebreak) return scores[i];
            return scores[i] === 4 ? "AD" : (scores[1-i] === 4 ? "" : pts[scores[i]]);
        };
        document.getElementById('p1-score').innerText = getL(0);
        document.getElementById('p2-score').innerText = getL(1);
        document.getElementById('drawer').classList.toggle('points-active', isGameActive);

        // Logo
        if (ui.logoSrc) document.getElementById('logo-img').src = ui.logoSrc;
        document.getElementById('logo-container').classList.toggle('active', showLogo && !!ui.logoSrc);

        // Panels
        document.getElementById('tiebreak-panel').classList.toggle('active', isTiebreak);
        document.getElementById('proset-indicator').classList.toggle('active', isProset);
        if (isProset) document.getElementById('proset-indicator').innerText = `PRO-SET ${prosetLimit}`;

        const isGP = !isAmerican && !isTiebreak && scores[0]===3 && scores[1]===3;
        document.getElementById('golden-panel').classList.toggle('active', isGP);

        const bn = document.getElementById('notif-panel');
        bn.innerText = notifText;
        bn.classList.toggle('active', !!notifText);

        // Serve dots & sets
        [0,1].forEach(p => {
            let h = "";
            setsHistory[p].forEach(g => h += `<div class="set-box-wrapper"><div class="set-box">${g}</div></div>`);
            h += `<div class="set-box-wrapper"><div class="set-box active">${games[p]}</div></div>`;
            document.getElementById(`p${p+1}-sets`).innerHTML = h;
            document.getElementById(`serve-${p}`).classList.toggle('active', server === p);
        });

        // Fullscreen alert
        const alertKey = state.alertKey || "";
        if (alertKey && alertKey !== lastAlertKey) {
            lastAlertKey = alertKey;
            const el = document.getElementById('scoreboard-fullscreen-alert');
            const txtEl = document.getElementById('fullscreen-alert-text');
            txtEl.innerText = state.alertText || "";
            let cls = "alert-text-content";
            if (state.alertMode === 'golden') cls += " alert-golden-theme";
            if (state.alertMode === 'tiebreak') cls += " alert-tiebreak-theme";
            txtEl.className = cls;
            el.classList.add('active');
            setTimeout(() => el.classList.remove('active'), 1500);
        }

        // Stats overlay
        if (state.showStats && !forceShowStats) {
            forceShowStats = true;
            updateStatsPanel(state);
            document.getElementById('tv-stats-overlay').classList.add('active');
            if (statsTimeout) clearTimeout(statsTimeout);
            statsTimeout = setTimeout(() => {
                forceShowStats = false;
                document.getElementById('tv-stats-overlay').classList.remove('active');
            }, 5000);
        }

        // Board visibility
        document.getElementById('scoreboard-anchor').classList.toggle('hidden-board', !showBoard);

        // Intro & finish
        document.getElementById('match-intro-overlay').classList.toggle('active', showIntro);
        if (showFinish) {
            updateFinishPanel(state);
            document.getElementById('match-finish-overlay').classList.add('active');
        } else {
            document.getElementById('match-finish-overlay').classList.remove('active');
        }

        // Stats logos
        if (ui.statLogoT) document.getElementById('stat-logo-t').src = ui.statLogoT;
        if (ui.statLogoF) document.getElementById('stat-logo-f').src = ui.statLogoF;
    }

    function updateStatsPanel(state) {
        const n1 = state.p1name || "Команда А";
        const n2 = state.p2name || "Команда Б";
        const setsHistory = state.setsHistory || [[],[]];
        const games = state.games || [0,0];
        const activeSetIndex = state.activeSetIndex || 0;
        const setTimes = state.setTimes || [0,0,0];

        let totalSets = Math.min(Math.max(activeSetIndex + 1, setsHistory[0].length, 1), 3);

        let headHtml = `<tr style="color:rgba(255,255,255,0.4); font-size:11px; text-transform:uppercase; height:25px;"><th style="text-align:left; padding-left:10px;">Команда</th>`;
        for(let s=1; s<=totalSets; s++) headHtml += `<th style="width:95px;">Set ${s}</th>`;
        headHtml += `</tr>`;
        document.getElementById('stats-table-head').innerHTML = headHtml;

        let s1=[], s2=[];
        for(let i=0; i<totalSets; i++) {
            s1.push(setsHistory[0][i] !== undefined ? setsHistory[0][i] : (activeSetIndex===i ? games[0] : '-'));
            s2.push(setsHistory[1][i] !== undefined ? setsHistory[1][i] : (activeSetIndex===i ? games[1] : '-'));
        }

        let bodyHtml = `<tr class="stats-row"><td class="stats-team-name">${n1}</td>`;
        for(let i=0; i<totalSets; i++) bodyHtml += `<td class="stats-set-column"><div class="stats-games-num">${s1[i]}</div><div class="stats-time-label">${s1[i] !== '-' ? `'${setTimes[i]}` : ''}</div></td>`;
        bodyHtml += `</tr><tr class="stats-row" style="border:none;"><td class="stats-team-name">${n2}</td>`;
        for(let i=0; i<totalSets; i++) bodyHtml += `<td class="stats-set-column"><div class="stats-games-num">${s2[i]}</div></td>`;
        bodyHtml += `</tr>`;
        document.getElementById('stats-table-body').innerHTML = bodyHtml;
    }

    function updateFinishPanel(state) {
        const n1 = state.p1name || "Команда А";
        const n2 = state.p2name || "Команда Б";
        const sh = state.setsHistory || [[],[]];
        let html = `<tr class="finish-row"><td class="finish-team">${n1}</td>`;
        for(let i=0;i<3;i++) html += `<td class="finish-cell-set"><span class="finish-box-score">${sh[0][i]!==undefined?sh[0][i]:'-'}</span></td>`;

        for(let i=0;i<3;i++) html += `<td class="finish-cell-set"><span class="finish-box-score">${sh[1][i]!==undefined?sh[1][i]:'-'}</span></td>`;
        html += `</tr>`;
        document.getElementById('finish-table-body').innerHTML = html;
    }

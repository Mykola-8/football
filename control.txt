<script type="module">
    import { initializeApp } from "https://www.gstatic.com/firebasejs/10.12.0/firebase-app.js";
    import { getDatabase, ref, set, update, get, onValue, serverTimestamp } from "https://www.gstatic.com/firebasejs/10.12.0/firebase-database.js";

    // ⬇️ ВСТАВТЕ ВАШІ ДАНІ FIREBASE ТУТ ⬇️
    const firebaseConfig = {
        apiKey: "ВСТАВТЕ_ВАШ_API_KEY",
        authDomain: "ВСТАВТЕ.firebaseapp.com",
        databaseURL: "https://ВСТАВТЕ-default-rtdb.firebaseio.com",
        projectId: "ВСТАВТЕ",
        storageBucket: "ВСТАВТЕ.appspot.com",
        messagingSenderId: "ВСТАВТЕ",
        appId: "ВСТАВТЕ"
    };

    const app = initializeApp(firebaseConfig);
    const db = getDatabase(app);
    const gameRef = ref(db, 'padel');

    // ─── LOCAL STATE ───────────────────────────────────────────
    const pts = [0, 15, 30, 40, "AD"];
    let state = {
        scores: [0,0], games: [0,0], setsHistory: [[],[]], server: 0,
        isGameActive: false, isTiebreak: false, isAmerican: false,
        isProset: false, deuceCycles: 0, setTimes: [0,0,0], activeSetIndex: 0,
        notifText: "", showFinish: false, showIntro: false, showLogo: false,
        showBoard: true, showStats: false, isGoldenPoint: true, prosetLimit: 8,
        americanLimit: 15, p1name: "Команда А", p2name: "Команда Б",
        courtName: "COURT 1", alertKey: "", alertText: "", alertMode: ""
    };
    let ui = {
        accentBlue: '#0055ff', ballColor: '#ccff00', teamNameColor: '#ffffff',
        scoreColor: '#ffffff', panelBg: 'rgba(60,60,60,0.7)', drawerBg: 'rgba(40,40,40,0.8)',
        popupBg: '#444444', popupText: '#ffffff', finishBg: 'rgba(255,255,255,0.03)',
        fontFamily: "'Oswald', sans-serif", sbW: '750px', sbH: '150px',
        sbRadius: '10px', sbSkew: '0deg', logoW: '35%', logoH: '50%',
        sbTop: 500, sbLeft: 100, logoSrc: '', statLogoT: '', statLogoF: ''
    };
    let historyStack = [];
    let gamePointCountP1 = 0, gamePointCountP2 = 0;
    let lastAlertKey = "";
    let setTimerInterval = null;

    // ─── FIREBASE PUSH ───────────────────────────────────────────
    function pushState() {
        set(gameRef, { state, ui }).catch(e => console.error(e));
    }

    // ─── CONNECTION STATUS ────────────────────────────────────────
    const connRef = ref(db, '.info/connected');
    onValue(connRef, (snap) => {
        const el = document.getElementById('conn-status');
        if (snap.val()) { el.textContent = 'ОНЛАЙН'; el.className = 'online'; }
        else { el.textContent = 'ОФЛАЙН'; el.className = 'offline'; }
    });

    // ─── SYNC FROM FIREBASE ON LOAD ──────────────────────────────
    onValue(gameRef, (snap) => {
        const d = snap.val();
        if (d) {
            if (d.state) state = { ...state, ...d.state };
            if (d.ui) ui = { ...ui, ...d.ui };
        }
        updateControlUI();
        renderMiniScoreboard();
    }, { onlyOnce: true });

    // ─── GAME LOGIC ───────────────────────────────────────────────
    function addPoint(p) {
        historyStack.push(JSON.parse(JSON.stringify(state)));
        if (!state.isGameActive) state.isGameActive = true;
        const o = 1 - p;

        if (state.isAmerican) {
            state.scores[p]++;
            if (state.scores[p] >= state.americanLimit) winGame(p);
            if ((state.scores[0]+state.scores[1])%2===0) state.server = 1-state.server;
        } else if (state.isTiebreak) {
            state.scores[p]++;
            if (state.scores[p]>=7 && state.scores[p]-state.scores[o]>=2) winGame(p);
            else if ((state.scores[0]+state.scores[1])%2===1) state.server = 1-state.server;
        } else {
            if (state.scores[p]===3 && state.scores[o]===3) {
                if (state.isGoldenPoint && state.deuceCycles>=2) winGame(p); else state.scores[p]=4;
            } else if (state.scores[p]===4) winGame(p);
            else if (state.scores[o]===4) { state.scores[o]=3; state.deuceCycles++; }
            else if (state.scores[p]===3) winGame(p);
            else state.scores[p]++;
        }
        computeNotif();
        pushState();
        updateControlUI();
    }

    function winGame(p) {
        state.games[p]++;
        state.scores = [0,0]; state.deuceCycles = 0; state.isGameActive = false;
        state.server = 1-state.server;
        gamePointCountP1 = 0; gamePointCountP2 = 0;

        const totalGames = state.games[0]+state.games[1];
        const isOddGame = totalGames%2!==0;
        const lim = state.prosetLimit || 8;

        let isSetFinished = false;
        if (state.isProset) {
            if (state.isTiebreak || (state.games[p]>=lim && state.games[p]-state.games[1-p]>=2)) isSetFinished = true;
            else if (state.games[p]===lim-1 && state.games[1-p]===lim-1) state.isTiebreak = true;
        } else if (!state.isAmerican) {
            if (state.isTiebreak || (state.games[p]>=6 && state.games[p]-state.games[1-p]>=2)) isSetFinished = true;
            else if (state.games[0]===6 && state.games[1]===6) state.isTiebreak = true;
        } else { isSetFinished = true; }

        if (isSetFinished) {
            state.setsHistory[0].push(state.games[0]);
            state.setsHistory[1].push(state.games[1]);
            state.games = [0,0]; state.isTiebreak = false;
            state.activeSetIndex = Math.min((state.activeSetIndex||0)+1, 2);
            state.showStats = true;
            setTimeout(() => { state.showStats = false; pushState(); }, 5100);
        } else if (isOddGame) {
            state.showStats = true;
            setTimeout(() => { state.showStats = false; pushState(); }, 5100);
        }
    }

    function computeNotif() {
        const s = state.scores; const g = state.games;
        let txt = ""; let alertText = ""; let alertMode = "";
        const deuceLim = state.isGoldenPoint;
        const isGP = !state.isAmerican && !state.isTiebreak && s[0]===3 && s[1]===3 && (!deuceLim || state.deuceCycles>=2);

        if (state.isGameActive && !state.isTiebreak && !state.isAmerican) {
            if (isGP) {
                txt = `GOLDEN POINT`;
                alertText = "GOLDEN POINT"; alertMode = "golden";
            } else {
                for (let i=0; i<2; i++) {
                    if (s[i]>=3 && s[i]>s[1-i]) {
                        const lim = state.isProset ? (state.prosetLimit||8)-1 : 5;
                        const isSetPt = g[i]>=lim;
                        const isBreak = state.server!==i;
                        if (isSetPt) { txt="SET POINT"; alertText="SET POINT"; alertMode="normal"; }
                        else {
                            if (i===0) { gamePointCountP1++; txt=(isBreak?`BREAK POINT #${gamePointCountP1}`:`GAME POINT #${gamePointCountP1}`); }
                            else { gamePointCountP2++; txt=(isBreak?`BREAK POINT #${gamePointCountP2}`:`GAME POINT #${gamePointCountP2}`); }
                            alertText = isBreak ? "BREAK POINT" : "GAME POINT"; alertMode = "normal";
                        }
                    }
                }
            }
        }
        if (state.isTiebreak && state.activeSetIndex !== (state._lastTBIdx||0)) {
            state._lastTBIdx = state.activeSetIndex;
            alertText = "TIE-BREAK"; alertMode = "tiebreak";
        }
        state.notifText = txt;
        if (alertText) {
            state.alertKey = alertText + "_" + Date.now();
            state.alertText = alertText; state.alertMode = alertMode;
        }
    }

    function doFault() {
        historyStack.push(JSON.parse(JSON.stringify(state)));
        addPoint(1-state.server);
    }

    function doUndo() {
        if (historyStack.length > 0) {
            state = historyStack.pop();
            pushState();
            updateControlUI();
        }
    }

    function setServer(p) {
        state.server = p;
        pushState();
        updateControlUI();
    }

    function sendToggle(key, val) {
        state[key] = val;
        pushState();
    }

    function sendMode() {
        state.isAmerican = document.getElementById('s-american').checked;

        state.americanLimit = parseInt(document.getElementById('s-american-limit').value)||15;
        state.scores = [0,0]; state.isGameActive = false;
        pushState();
    }
